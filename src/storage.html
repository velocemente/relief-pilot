function getFlights() { try { return JSON.parse(localStorage.getItem('rp_flights') || '[]'); } catch(e) { return []; } } function putFlights(arr) { try { localStorage.setItem('rp_flights', JSON.stringify(arr.slice(0,25))); } catch(e) {} } function saveFlight() { if (!S.fltNum && !S.date) return; const flights = getFlights(); const id = 'f_' + Date.now(); const idx = flights.findIndex(f => f.fltNum === S.fltNum && f.date === S.date); const entry = { id, fltNum:S.fltNum, date:S.date, at:new Date().toISOString(), state:{...S}, raw:pdfRawText.substring(0,10000), aptData: JSON.parse(JSON.stringify(airportData)) }; const stableId = idx >= 0 ? flights[idx].id : entry.id; if (idx >= 0) { flights[idx] = {...entry, id: stableId}; } else { flights.unshift(entry); } putFlights(flights); // Persist PDF binary to IndexedDB (fire-and-forget — non-blocking) if (pdfArrayBuffer) _pdfDb.set(stableId, pdfArrayBuffer).catch(() => {}); } function deleteFlight(id) { // Block deletion if FCIR/ASAP is active — must be turned off first if (_fcirActive) { showToast('Turn off FCIR/ASAP toggle before deleting this flight.', 'error'); return; } putFlights(getFlights().filter(f => f.id !== id)); _pdfDb.del(id); // remove PDF binary from IndexedDB _clearResumeView(); renderFlightList(); } function clearOFPPanel() { // Wipe the OFP viewer so the next loadOFPPanel() call renders fresh content. // Must be called whenever a new flight is loaded or a new PDF is imported. _pdfDocProxy = null; const c = document.getElementById('ofpViewerContainer'); if (c) c.innerHTML = ''; const ctr = document.getElementById('ofpPageCounter'); if (ctr) ctr.textContent = ''; } function loadFlight(entry) { // Always start from a clean slate — prevents any residual state from a // previously viewed flight bleeding into the loaded one. resetStateForNewFlight(); // Restore state from saved entry Object.assign(S, entry.state || {}); pdfRawText = entry.raw || ''; pdfArrayBuffer = null; // start null; async restore from IndexedDB below clearOFPPanel(); // wipe any canvas pages from the previously viewed flight _fcirActive = !!(S.fcirAsap); // Restore airport wx/notam data from saved entry airportData = entry.aptData ? JSON.parse(JSON.stringify(entry.aptData)) : {}; // Restore the flight date for timezone calculations window._flightDateISO = ''; if (S.date) { // Re-derive ISO date from stored date string (e.g. "06 MAR 26") try { const MONTHS = {JAN:0,FEB:1,MAR:2,APR:3,MAY:4,JUN:5,JUL:6,AUG:7,SEP:8,OCT:9,NOV:10,DEC:11}; const dm = S.date.match(/(\d{2})\s+([A-Z]{3})\s+(\d{2})/); if (dm) { const d = new Date(Date.UTC(2000+parseInt(dm[3]), MONTHS[dm[2]], parseInt(dm[1]))); window._flightDateISO = d.toISOString().slice(0,10); } } catch(e) {} } // Async: restore PDF binary from IndexedDB so OFP tab shows pages, not text if (entry.id) { _pdfDb.get(entry.id).then(buf => { if (buf) { pdfArrayBuffer = buf; // If the OFP panel is currently visible, re-render it with the PDF const ofpPanel = document.getElementById('fpanel-ofp'); if (ofpPanel && ofpPanel.classList.contains('active')) { clearOFPPanel(); loadOFPPanel(); } } }).catch(() => {}); } renderNote(); push('screen-flight'); const tog = document.getElementById('fcirToggle'); if (tog) tog.checked = _fcirActive; const lbl = document.getElementById('fcirLabel'); if (lbl) lbl.classList.toggle('active', _fcirActive); _saveResumeView(); } function renderFlightList() { const flights = getFlights(); const body = document.getElementById('flightListBody'); const empty = document.getElementById('emptyFlights'); body.innerHTML = ''; if (!flights.length) { empty.style.display = 'flex'; return; } empty.style.display = 'none'; flights.forEach(f => { const saved = new Date(f.at).toLocaleDateString('en-US', {month:'short', day:'numeric', year:'numeric'}); const isFCIR = !!(f.state && f.state.fcirAsap); const cell = document.createElement('div'); cell.className = 'section'; cell.innerHTML = `
${esc(f.fltNum||'—')}
${esc(f.date||'')} · Saved ${saved}${isFCIR ? ' · FCIR/ASAP' : ''}
`; cell.querySelector('.cell').addEventListener('click', () => loadFlight(f)); body.appendChild(cell); }); }
